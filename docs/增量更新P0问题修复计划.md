## 2. 问题1: 并发安全风险

### 2.1 问题详情

**代码位置**: `src/data_interface/increment_manager.rs:697-723`

**当前代码**:
```rust
// ❌ 问题代码
if let Some(mut old) = self.watcher.headers.get_mut(path) {
    dbg!(path);
    dbg!(new_header.latest_ses_data.sesno);
    #[cfg(feature = "web_server")]
    let prev_sesno = old.latest_ses_data.sesno;
    let new_sesno = new_header.latest_ses_data.sesno;

    // ⚠️ 持有锁期间执行数据库查询（长时间操作）
    let db_num = new_header.pdms_header.db_num;
    let db_latest_sesno =
        match Self::query_latest_sesno_by_dbnum(db_num as _).await {
            Ok(sesno) => sesno,
            Err(e) => {
                println!("查询数据库最新sesno失败: {:?}", e);
                continue;
            }
        };

    // 未发生修改，直接跳过
    if db_latest_sesno as i32 == new_sesno {
        continue;
    }
    // 比如给出准确的范围next_sesno..=end_sesno
    params.insert(
        path.clone(),
        (new_header.clone(), (db_latest_sesno as i32 + 1)..=new_sesno),
    );
}
```

**问题分析**:

1. **锁持有时间过长**:
   - `get_mut()` 获取可变引用时持有 `DashMap` 的分片锁
   - 持有锁期间执行异步数据库查询（可能耗时数百毫秒）
   - 持有锁期间执行 `params.insert()`（涉及克隆操作）

2. **阻塞其他操作**:
   ```
   文件A变化 → 获取锁 → 查询数据库(200ms) → 释放锁
                          ↑
                          文件B变化 → 等待锁 ⏳
   ```

3. **潜在死锁风险**:
   - 如果数据库查询内部也访问 `headers`，可能形成循环等待
   - 虽然当前代码未发现，但架构存在风险

### 2.2 修复方案

#### 方案A: 先读后写（推荐）

**实现代码**:

**风险分析**:

1. **临时故障导致永久丢失**:
   ```

   ```

### 3.2 修复方案

#### 核心设计: 失败任务队列 + 重试机制

**架构设计**:

```
┌─────────────────────────────────────────────────────────┐
│                    错误恢复架构                           │
└─────────────────────────────────────────────────────────┘

增量处理流程
     │
     ├─ 成功 ✅ → 正常完成
     │
     └─ 失败 ❌ → 失败任务队列
                      │
                      ├─ 内存队列 (临时存储)
                      │   └─ Vec<FailedTask>
                      │
                      ├─ 磁盘持久化 (断电保护)
                      │   └─ assets/failed_tasks.json
                      │
                      └─ 后台重试线程
                          │
                          ├─ 每60秒扫描一次
                          ├─ 按优先级排序
                          ├─ 指数退避重试
                          │   └─ 1分钟 → 2分钟 → 4分钟 → ...
                          │
                          ├─ 最大重试次数: 5次
                          │
                          └─ 超过阈值 → 告警通知
```

**数据结构定义**:

```rust
use std::path::PathBuf;
use std::time::SystemTime;
use serde::{Deserialize, Serialize};

/// 失败任务类型
#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum FailedTaskType {
    /// 数据库查询失败
    DatabaseQuery {
        dbnum: u32,
        operation: String,  // "query_sesno", "update_elements", etc.
    },
    /// CBA压缩失败
    Compression {
        input_path: PathBuf,
        output_path: PathBuf,
    },
    /// 增量更新失败
    IncrementUpdate {
        path: PathBuf,
        sesno_range: String,  // "12341..=12350"
    },
    /// MQTT推送失败
    MqttPublish {
        topic: String,
        payload_summary: String,
    },
}

/// 失败任务记录
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct FailedTask {
    /// 任务ID (UUID)
    pub id: String,

    /// 任务类型
    pub task_type: FailedTaskType,

    /// 错误信息
    pub error: String,

    /// 错误堆栈 (如果有)
    pub error_trace: Option<String>,

    /// 重试次数
    pub retry_count: u32,

    /// 最大重试次数
    pub max_retries: u32,

    /// 首次失败时间
    pub first_failed_at: SystemTime,

    /// 最后重试时间
    pub last_retry_at: Option<SystemTime>,

    /// 下次重试时间 (指数退避)
    pub next_retry_at: SystemTime,

    /// 优先级 (1-10, 10最高)
    pub priority: u8,

    /// 任务元数据 (JSON)
    pub metadata: serde_json::Value,
}

impl FailedTask {
    /// 创建新的失败任务
    pub fn new(task_type: FailedTaskType, error: impl ToString) -> Self {
        let now = SystemTime::now();
        Self {
            id: uuid::Uuid::new_v4().to_string(),
            task_type,
            error: error.to_string(),
            error_trace: None,
            retry_count: 0,
            max_retries: 5,
            first_failed_at: now,
            last_retry_at: None,
            next_retry_at: now + Duration::from_secs(60), // 1分钟后重试
            priority: 5,
            metadata: serde_json::Value::Null,
        }
    }

    /// 计算下次重试时间 (指数退避)
    pub fn schedule_next_retry(&mut self) {
        self.retry_count += 1;
        self.last_retry_at = Some(SystemTime::now());

        // 指数退避: 1min, 2min, 4min, 8min, 16min
        let delay_secs = 60 * (2u64.pow(self.retry_count.min(4)));
        self.next_retry_at = SystemTime::now() + Duration::from_secs(delay_secs);
    }

    /// 判断是否应该重试
    pub fn should_retry(&self) -> bool {
        self.retry_count < self.max_retries
            && SystemTime::now() >= self.next_retry_at
    }

    /// 判断是否已达到最大重试次数
    pub fn is_exhausted(&self) -> bool {
        self.retry_count >= self.max_retries
    }
}

/// 失败任务队列管理器
#[derive(Clone)]
pub struct FailedTaskQueue {
    /// 内存队列
    tasks: Arc<RwLock<Vec<FailedTask>>>,

    /// 持久化文件路径
    persist_path: PathBuf,
}

impl FailedTaskQueue {
    pub fn new(persist_path: PathBuf) -> Self {
        let tasks = Self::load_from_disk(&persist_path).unwrap_or_default();
        Self {
            tasks: Arc::new(RwLock::new(tasks)),
            persist_path,
        }
    }

    /// 添加失败任务
    pub async fn push(&self, task: FailedTask) {
        let mut tasks = self.tasks.write().await;
        tasks.push(task);
        drop(tasks);

        // 异步持久化
        self.persist().await;
    }

    /// 获取待重试的任务
    pub async fn get_pending_tasks(&self) -> Vec<FailedTask> {
        let tasks = self.tasks.read().await;
        tasks
            .iter()
            .filter(|t| t.should_retry())
            .cloned()
            .collect()
    }

    /// 移除任务 (成功后)
    pub async fn remove(&self, task_id: &str) {
        let mut tasks = self.tasks.write().await;
        tasks.retain(|t| t.id != task_id);
        drop(tasks);

        self.persist().await;
    }

    /// 更新任务 (重试后)
    pub async fn update(&self, task: FailedTask) {
        let mut tasks = self.tasks.write().await;
        if let Some(existing) = tasks.iter_mut().find(|t| t.id == task.id) {
            *existing = task;
        }
        drop(tasks);

        self.persist().await;
    }

    /// 获取已达到最大重试次数的任务 (用于告警)
    pub async fn get_exhausted_tasks(&self) -> Vec<FailedTask> {
        let tasks = self.tasks.read().await;
        tasks.iter().filter(|t| t.is_exhausted()).cloned().collect()
    }

    /// 持久化到磁盘
    async fn persist(&self) {
        let tasks = self.tasks.read().await;
        let json = serde_json::to_string_pretty(&*tasks).unwrap();

        if let Err(e) = tokio::fs::write(&self.persist_path, json).await {
            eprintln!("持久化失败任务失败: {:?}", e);
        }
    }

    /// 从磁盘加载
    fn load_from_disk(path: &Path) -> Option<Vec<FailedTask>> {
        let content = std::fs::read_to_string(path).ok()?;
        serde_json::from_str(&content).ok()
    }
}
```

```rust
impl AiosDBManager {
    /// 失败任务队列 (添加到结构体中)
    pub failed_queue: FailedTaskQueue,

    /// 查询会话号 (带错误恢复)
    async fn query_latest_sesno_with_retry(&self, dbnum: u32) -> anyhow::Result<u32> {
        match Self::query_latest_sesno_by_dbnum(dbnum).await {
            Ok(sesno) => Ok(sesno),
            Err(e) => {
                // 记录失败任务
                let task = FailedTask::new(
                    FailedTaskType::DatabaseQuery {
                        dbnum,
                        operation: "query_latest_sesno".to_string(),
                    },
                    &e,
                );
                self.failed_queue.push(task).await;

                // 返回错误
                Err(e)
            }
        }
    }

    /// CBA压缩 (带错误恢复)
    async fn compress_with_retry(
        &self,
        opts: CompressOptions,
    ) -> anyhow::Result<String> {
        match execute_compress(opts.clone()).await {
            Ok(hash) => Ok(hash.to_string()),
            Err(e) => {
                // 记录失败任务
                let task = FailedTask::new(
                    FailedTaskType::Compression {
                        input_path: opts.input.clone(),
                        output_path: opts.output.clone(),
                    },
                    &e,
                );
                self.failed_queue.push(task).await;

                Err(e)
            }
        }
    }

    /// 后台重试线程
    pub async fn start_retry_worker(&self) {
        let queue = self.failed_queue.clone();
        let mgr = self.clone();

        tokio::spawn(async move {
            loop {
                tokio::time::sleep(Duration::from_secs(60)).await;

                // 获取待重试的任务
                let pending = queue.get_pending_tasks().await;

                for task in pending {
                    info!(
                        task_id = %task.id,
                        retry_count = task.retry_count,
                        "开始重试失败任务"
                    );

                    let result = mgr.retry_failed_task(&task).await;

                    match result {
                        Ok(()) => {
                            // 重试成功，移除任务
                            info!(task_id = %task.id, "重试成功");
                            queue.remove(&task.id).await;
                        }
                        Err(e) => {
                            // 重试失败，更新任务
                            let mut updated_task = task.clone();
                            updated_task.error = e.to_string();
                            updated_task.schedule_next_retry();

                            warn!(
                                task_id = %task.id,
                                retry_count = updated_task.retry_count,
                                next_retry = ?updated_task.next_retry_at,
                                error = ?e,
                                "重试失败"
                            );

                            queue.update(updated_task).await;
                        }
                    }
                }

                // 检查已耗尽重试次数的任务
                let exhausted = queue.get_exhausted_tasks().await;
                for task in exhausted {
                    error!(
                        task_id = %task.id,
                        task_type = ?task.task_type,
                        error = %task.error,
                        "任务已达到最大重试次数"
                    );

                    // TODO: 发送告警通知
                    // alert_manager.send_alert(
                    //     AlertLevel::Critical,
                    //     format!("增量更新任务失败: {}", task.id)
                    // );
                }
            }
        });
    }

    /// 重试失败任务
    async fn retry_failed_task(&self, task: &FailedTask) -> anyhow::Result<()> {
        match &task.task_type {
            FailedTaskType::DatabaseQuery { dbnum, operation } => {
                Self::query_latest_sesno_by_dbnum(*dbnum).await?;

                    input_path.clone(),

            }
            FailedTaskType::IncrementUpdate { path, sesno_range } => {
                // TODO: 重新执行增量更新
                Ok(())
            }
            FailedTaskType::MqttPublish { topic, payload_summary } => {
                // TODO: 重新推送MQTT
                Ok(())
            }
        }
    }
}
```

```rust
    let tasks: Vec<_> = (0..num_files)

    }

}
```

```rust
#[tokio::test]
async fn test_retry_backoff() {
    let mut task = FailedTask::new(
        FailedTaskType::DatabaseQuery {
            dbnum: 1001,
            operation: "test".to_string(),
        },
        "Test error",
    );

    let initial_time = task.next_retry_at;

    // 第1次重试
    task.schedule_next_retry();
    let delay1 = task.next_retry_at.duration_since(SystemTime::now()).unwrap();
    assert!(delay1.as_secs() >= 60 && delay1.as_secs() <= 120); // ~1分钟

    // 第2次重试
    task.schedule_next_retry();
    let delay2 = task.next_retry_at.duration_since(SystemTime::now()).unwrap();
    assert!(delay2.as_secs() >= 120 && delay2.as_secs() <= 180); // ~2分钟

    // 第3次重试
    task.schedule_next_retry();
    let delay3 = task.next_retry_at.duration_since(SystemTime::now()).unwrap();
    assert!(delay3.as_secs() >= 240 && delay3.as_secs() <= 300); // ~4分钟
}
```
